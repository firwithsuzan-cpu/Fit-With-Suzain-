    /*
      PRODUCTION SETUP:
      1. Replace this URL with your real HTTPS lead-capture endpoint.
      2. Your server/provider should validate inputs, record consent and
         deliver the guide/email through a trusted email service.
      3. Never put email-service API keys or secret credentials in this file.
    */
    const WEBHOOK_URL = "YOUR_WEBHOOK_URL";

    const form = document.getElementById("leadForm");
    const submitButton = document.getElementById("submitButton");
    const status = document.getElementById("formStatus");

    document.getElementById("year").textContent = new Date().getFullYear();

    function showError(id, message) {
      const element = document.getElementById(id);
      element.textContent = message;
      element.classList.toggle("hidden", !message);
    }

    function validEmail(email) {
      return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email);
    }

    form.addEventListener("submit", async function (event) {
      event.preventDefault();

      const firstName = document.getElementById("firstName").value.trim();
      const email = document.getElementById("email").value.trim();
      const consent = document.getElementById("consent").checked;

      showError("nameError", "");
      showError("emailError", "");
      showError("consentError", "");
      status.textContent = "";

      let valid = true;

      if (!firstName || firstName.length > 80) {
        showError("nameError", "Please enter your first name.");
        valid = false;
      }

      if (!validEmail(email) || email.length > 254) {
        showError("emailError", "Please enter a valid email address.");
        valid = false;
      }

      if (!consent) {
        showError("consentError", "Please agree to receive the guide by email.");
        valid = false;
      }

      if (!valid) return;

      submitButton.disabled = true;
      submitButton.textContent = "Please wait...";
      status.textContent = "Submitting your details securely...";

      const payload = {
        firstName,
        email,
        consent: true,
        consentTextVersion: "email-consent-v1",
        source: "fit-with-suzain-joint-guide",
        submittedAt: new Date().toISOString()
      };

      try {
        if (!WEBHOOK_URL || WEBHOOK_URL === "YOUR_WEBHOOK_URL") {
          /*
            DEMO ONLY:
            This temporary browser storage does NOT send an email or save a
            lead to your business database. Configure a real webhook before
            using this page for live traffic.
          */
          sessionStorage.setItem("fitSuzainDemoLead", JSON.stringify(payload));
          window.location.assign("thank-you.html?demo=1");
          return;
        }

        const response = await fetch(WEBHOOK_URL, {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify(payload)
        });

        if (!response.ok) {
          throw new Error("Lead submission failed.");
        }

        // Only redirect after the endpoint acknowledges the submission.
        window.location.assign("thank-you.html");
      } catch (error) {
        console.error("Lead submission error:", error);
        status.textContent =
          "We couldn't submit your details. Please check your connection and try again.";
        submitButton.disabled = false;
        submitButton.textContent =
          "Get Instant Access to 7-Day Protocol (Free PDF)";
      }
    });
  </script>
